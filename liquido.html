/* SCRIPT CORREGIDO PARA FILTRADO AGRUPADO POR MES */ function parseRowDate(str) { if (!str) return NaN; // Limpia espacios y caracteres raros const cleanStr = str.trim(); // Caso DD/MM/YYYY o D/M/YY const p = cleanStr.split('/'); if (p.length === 3) { const year = p[2].length === 2 ? parseInt('20' + p[2]) : parseInt(p[2]); return new Date(year, parseInt(p[1]) - 1, parseInt(p[0])); } // Caso YYYY-MM-DD const d = cleanStr.split('-'); if (d.length === 3) { return new Date(parseInt(d[0]), parseInt(d[1]) - 1, parseInt(d[2])); } return NaN; } function buildMonthFilters() { const monthsDiv = document.getElementById('filterMonths'); monthsDiv.innerHTML = ''; const uniqueMonths = new Set(); const targetData = currentModule === 'GNC' ? rawGncData : rawLiqData; targetData.forEach(r => { const d = parseRowDate(r[0]); if (!isNaN(d)) { // Agrupa correctamente por Mes y Año (ej: "May 26", "Jun 26") const label = d.toLocaleDateString('es-AR', { month: 'short', year: '2-digit' }); uniqueMonths.add(label.replace('.', '')); // Elimina puntos si los agrega el formato local } }); // Convertir a Array y ordenar de forma cronológica aproximada const sortedMonths = Array.from(uniqueMonths); // Botón "Todos" (Como en la segunda imagen, para que no inicie vacío) const btnAll = document.createElement('button'); btnAll.className = `pill px-3 py-1 rounded-full ${activeFilters.month === 'Todos' ? 'active' : ''}`; btnAll.textContent = "Todos"; btnAll.onclick = () => { selectFilter('month', 'Todos'); }; monthsDiv.appendChild(btnAll); // Agregar el resto de los meses limpios sortedMonths.forEach(m => { const btn = document.createElement('button'); btn.className = `pill px-3 py-1 rounded-full ${activeFilters.month === m ? 'active' : ''}`; btn.textContent = m.toUpperCase(); btn.onclick = () => { selectFilter('month', m); }; monthsDiv.appendChild(btn); }); } function updateDashboard() { const isGnc = currentModule === 'GNC'; const dataSrc = isGnc ? rawGncData : rawLiqData; // Forzar que si es la primera carga o cambia de pestaña, empiece mostrando "Todos" para ver los datos if (!activeFilters.month) { activeFilters.month = 'Todos'; } // 1. Filtrado Base por Tiempo let baseRows = dataSrc.filter(r => { const d = parseRowDate(r[0]); if (isNaN(d)) return false; if (activeFilters.month === 'Todos') { return true; } else { const label = d.toLocaleDateString('es-AR', { month: 'short', year: '2-digit' }).replace('.', '').toUpperCase(); return label === activeFilters.month.toUpperCase(); } }); // Construir filtros de playeros en base a este set buildWorkerFilters(baseRows); // 2. Filtros operativos aplicados.`; // Renderizar módulos correspondientes if (isGnc) { renderGncDashboard(displayRows); } else { renderLiquidosDashboard(displayRows); } } // Asegurarse de inicializar el filtro por defecto en "Todos" al arrancar activeFilters.month = 'Todos';
